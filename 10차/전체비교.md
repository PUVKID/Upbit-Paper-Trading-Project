+## 현재 상태
+
+10차는 9차 이후 승인한 모델 개선을 유한 연구로 실행한 차수입니다. **실제 실행은 종료 코드0, 적합30회·완료30회, 전 모델 `cuda:0` 및 저장/재로드 검증 완료입니다. 신규 후보는 후속 평가의 거래 수·순수익 조건을 충족하지 못해 미채택했습니다.** 적합 성공과 거래 우위는 다릅니다. 최종 검증과 원격 반영은 별도 영수증으로 관리하며 GPU 성과와 구분합니다. 수익·정확도100%·낙폭10% 이내·운용 적용을 보장하지 않습니다.

+## 실제 연구 결과와 미채택
+
+[공개 집계 JSON](실험결과.json)의 `review.eligible=false`, `promotion=false`, `candidate_ready=false`를 확인했습니다. 기각 사유는 fold별 거래 부족(`insufficient_fold_trades`), 총 청산 거래 부족(`insufficient_total_closed_trades`), 양수가 아닌 fold 수익(`non_positive_fold_return`)입니다. 두 fold 모두 최종 선정 정책은 현금 보유입니다.

+def return_signals(values, policy, factor_ratio):
+    import numpy as np
+    if policy['family'] == 'cash':
+        return np.ones(len(values), dtype=int)

+        raise ValueError('비용 스트레스 비율 위반')

+    spread = (values['q90']-values['q10']).to_numpy()*factor_ratio
+    finite = np.isfinite(values[['mean', 'q10', 'q90']].to_numpy()).all(axis=1)
+    score = mean-policy['penalty']*spread
+    return np.where(finite & (spread >= 0) & (mean > 0) & (score > policy['minimum']), 2, 1)

+def _reconstruct(state, result):
+    if not isinstance(state, dict) or state.get("paper_only") is not True:

+def safe_path(value, boundary, root):
+    """resolve로 링크를 숨기지 않고 모든 기존 경로 요소를 검사합니다."""
+    value, root, boundary = Path(value), Path(root).absolute(), Path(boundary).absolute()
+    path = value if value.is_absolute() else root / value
+    if ".." in path.parts or not path.is_relative_to(boundary) or path == boundary:
+        raise ValueError("unsafe_path")
+    for part in path.parts[1:]:
+        if ":" in part:
+            raise ValueError("unsafe_path")
+    for current in reversed((path, *path.parents)):
+        try:
+            info = current.lstat()
+        except FileNotFoundError:
+            continue
+        if stat.S_ISLNK(info.st_mode) or getattr(info, "st_file_attributes", 0) & 0x400:
+            raise ValueError("linked_path")
+        if current != path and not stat.S_ISDIR(info.st_mode):
+            raise ValueError("invalid_parent")
+        if current == path and (not stat.S_ISREG(info.st_mode) or info.st_nlink != 1):
+            raise ValueError("invalid_file")
+    return path

+def parse_state(raw):
+    def pairs(items):

+            result[key] = value
+        return result
+    def constant(value):
+        raise ValueError("nonfinite_json")
+    return json.loads(raw, object_pairs_hook=pairs, parse_constant=constant)

+def _read_snapshot(path, root):
+    path = safe_path(path, root, root)

+        before = os.fstat(stream.fileno())
+        raw = stream.read(MAX_BYTES + 1)

+    identity = lambda info: (info.st_dev, info.st_ino, info.st_size, info.st_mtime_ns)
+    if identity(before) != identity(after) or identity(after) != identity(current):
+        raise ValueError("snapshot_changed")
+    if len(raw) > MAX_BYTES:
+        raise ValueError("input_too_large")
+    return raw

+def stable_read(path, root):
+    """변하는 장부를 잠그거나 복제하지 않고 동일한 두 읽기를 유한 확인합니다."""

+            first = _read_snapshot(path, root)

+        if first == second:
